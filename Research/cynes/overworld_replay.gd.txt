## The remake's side of Research/cynes/overworld_checks.py: replays a
## recording's buttons on world 1's overworld and writes what the remake did,
## frame by frame, for comparing. To use it, copy it into the project as a
## script on a bare Node in a temporary scene and run
##   godot --headless --fixed-fps 60 res://tmp.tscn -- recording.json out.json [world 0-4]
## The original acts on a button one frame after it is set, and drops a
## frame now and then (its frame counter does not move): both are allowed for.
extends Node
const BITS := {1: "move_right", 2: "move_left", 4: "move_down", 8: "move_up"}
func _ready() -> void:
	var args := OS.get_cmdline_user_args()
	var rows: Array = JSON.parse_string(FileAccess.get_file_as_string(args[0]))
	var sewer = load("res://Scenes/Levels/Area1Sewer.tscn").instantiate()
	add_child(sewer)
	await get_tree().physics_frame
	if args.size() > 2:
		sewer.load_overworld(int(args[2]))
	sewer.frame_clock = int(rows[0].fc) - 1
	var w: OverworldTurtle = sewer.walker
	w.position = Vector2(rows[0].tx, rows[0].ty)
	w.stand()
	sewer._place_enemies()
	sewer.set_overworld_screen(Vector2(rows[0].cam[0], rows[0].cam[1]))
	w._frame = 0
	var out := []
	var last_fc := -1
	var held := 0
	for row: Dictionary in rows:
		var lag := int(row.fc) == last_fc
		last_fc = int(row.fc)
		if lag:
			out.append(out[-1])
			sewer.frame_clock = int(row.fc)
			continue
		for bit: int in BITS:
			if held & bit:
				Input.action_press(BITS[bit])
			else:
				Input.action_release(BITS[bit])
		held = int(row.btn)
		w.safe_updates = 1000   # the recordings were made with the turtle kept safe
		w.visible = true
		await get_tree().physics_frame
		var things := []
		for e in sewer.enemies.get_children():
			if e is OverworldEnemy and not e.is_queued_for_deletion():
				things.append([e.kind, e.position.x, e.position.y, e.picture_name, e.facing_left, e.speed.x, e.speed.y])
		out.append({"t": [w.position.x, w.position.y], "cam": [sewer.overworld_screen.x, sewer.overworld_screen.y], "e": things})
	var file := FileAccess.open(args[1], FileAccess.WRITE)
	file.store_string(JSON.stringify(out))
	file.close()
	get_tree().quit()
